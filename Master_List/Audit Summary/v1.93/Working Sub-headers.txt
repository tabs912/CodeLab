function createOrRefreshBaseTemplate_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var baseSheet = ss.getSheetByName("RFF_BASE_TEMPLATE");
  
  if (!baseSheet) {
    baseSheet = ss.insertSheet("RFF_BASE_TEMPLATE");
  } else {
    baseSheet.clear();
    baseSheet.clearFormats();
  }

  var freezeRows = getDashboardSetting_("Freeze Rows", GLOBAL_DEFAULTS.FREEZE_ROWS);
  var freezeCols = getDashboardSetting_("Freeze Columns", GLOBAL_DEFAULTS.FREEZE_COLS);
  baseSheet.setFrozenRows(freezeRows);
  baseSheet.setFrozenColumns(freezeCols);

  var baseCols = 50;
  resizeSheetGrid_(baseSheet, 100, baseCols);

  var fullRange = baseSheet.getRange(1, 1, 100, baseCols);
  fullRange.setFontFamily(getDashboardSetting_("Standard Font", GLOBAL_DEFAULTS.STANDARD_FONT));
  fullRange.setFontSize(getDashboardSetting_("Standard Font Size", GLOBAL_DEFAULTS.STANDARD_FONT_SIZE));
  fullRange.setFontColor(getDashboardSetting_("Standard Font Color", GLOBAL_DEFAULTS.STANDARD_FONT_COLOR));
  fullRange.setHorizontalAlignment("left");
  fullRange.setVerticalAlignment("middle");
  fullRange.setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP);
  fullRange.setNumberFormat(GLOBAL_DEFAULTS.DEFAULT_TEXT_FORMAT);
  fullRange.setBorder(true, true, true, true, true, true, GLOBAL_DEFAULTS.BORDER_COLOR, GLOBAL_DEFAULTS.BORDER_STYLE);

  baseSheet.setRowHeight(1, getDashboardSetting_("Row 1 (Title) Height", GLOBAL_DEFAULTS.ROW_HEIGHT_TITLE));
  baseSheet.setRowHeight(2, getDashboardSetting_("Row 2 Height", GLOBAL_DEFAULTS.ROW_HEIGHT_DATE));
  baseSheet.setRowHeight(3, getDashboardSetting_("Row 3 Height", GLOBAL_DEFAULTS.ROW_HEIGHT_SPACER));
  
  var dataRowHeight = getDashboardSetting_("Default Data Row Height", GLOBAL_DEFAULTS.ROW_HEIGHT_DATA);
  for (var r = 4; r <= 100; r++) {
    baseSheet.setRowHeight(r, dataRowHeight);
  }

  var defaultWidth = getDashboardSetting_("Default Column Width", GLOBAL_DEFAULTS.DEFAULT_COL_WIDTH);
  for (var c = 1; c <= baseCols; c++) {
    baseSheet.setColumnWidth(c, defaultWidth);
  }

  var cellA1 = baseSheet.getRange("A1");
  cellA1.setFontSize(getDashboardSetting_("A1 (Title) Font Size", 14));
  cellA1.setFontWeight("bold");
  cellA1.setWrapStrategy(SpreadsheetApp.WrapStrategy.OVERFLOW);

  var cellC1 = baseSheet.getRange("C1");
  cellC1.setFontSize(getDashboardSetting_("C1 (Title) Font Size", 5));
  cellC1.setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);

  baseSheet.getRange(4, 1, 1, baseCols).setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);

  baseSheet.hideSheet();
  return baseSheet;
}

function createTemplateFromBase_(targetName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const existing = ss.getSheetByName(targetName);
  
  if (existing) {
    ss.deleteSheet(existing);
    SpreadsheetApp.flush();
  }
  
  const base = ss.getSheetByName("RFF_BASE_TEMPLATE") || createOrRefreshBaseTemplate_();
  const newSheet = base.copyTo(ss);
  newSheet.setName(targetName);
  newSheet.showSheet();
  SpreadsheetApp.flush();
  
  return newSheet;
}

function buildGeneralTemplate_(config) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sysName = config.systemSheetName;
  var templateName = config.templateName || ("Template - " + sysName);

  var templateSheet = createTemplateFromBase_(templateName);

  positionSheetBySectionFRank_(templateSheet, config.rank);

  var targetCols = Math.max(
    config.columnCount || 0,
    config.headers ? config.headers.length : 0,
    10
  );
  var targetRows = config.minimumRows || 100;
  resizeSheetGrid_(templateSheet, targetRows, targetCols);

  var theme = config.theme || getTemplateTheme_(sysName);

  if (config.usesAlternatingColors !== false && sysName !== "Format Dashboard") {
    applyNativeBandingSafe_(templateSheet, 5, targetCols, targetRows, theme.level4, theme.level5);
  }

  applyTitleRows_(templateSheet, config, theme);

  if (config.usesSubheaders && config.subheaders && config.subheaders.length > 0) {
    applySubheaderBlocks_(templateSheet, config.subheaders, theme, sysName);
  }

  applyColumnWidthsEngine_(templateSheet, config);

  if (theme.level1) templateSheet.setTabColor(theme.level1);
  writeTemplateMetadata_(templateSheet, config, targetCols);
  storeTemplateFormatSignature_(templateName, config);

  templateSheet.hideSheet();
  return templateSheet;
}

/**
 * Title Rows Renderer for Template Molds.
 * NOTE: Dynamic timestamps are NOT stamped on template molds.
 */
function applyTitleRows_(sheet, config, theme) {
  var sysName = config.systemSheetName || "";
  var sheetType = config.sheetType || "DATA";
  var fullGridCols = sheet.getMaxColumns();

  var titleCell = sheet.getRange("A1");
  titleCell.setValue(config.reportTitle || sysName);
  titleCell.setFontSize(14);
  titleCell.setFontWeight("bold");

  if (theme.level3) {
    sheet.getRange(1, 1, 2, fullGridCols).setBackground(theme.level3);
  }
  if (theme.titleFontColor) {
    titleCell.setFontColor(theme.titleFontColor);
  }

  if (sheetType === "DATA") {
    sheet.getRange("A2").setValue("Date");
    sheet.getRange("C2").setValue("to");
  } else if (sysName.indexOf("Timing") !== -1 || sysName.indexOf("Quality") !== -1) {
    sheet.getRange("A2").setValue("Date Created");
  } else if (sysName === "Index") {
    sheet.getRange("A2").setValue("Last Update");
  }

  if (theme.level1) {
    sheet.getRange(3, 1, 1, fullGridCols).setBackground(theme.level1);
  }

  if (theme.level2) {
    sheet.getRange(4, 1, 1, fullGridCols).setBackground(theme.level2);
  }

  if (!config.usesSubheaders && config.headers && config.headers.length > 0) {
    var headerTextRange = sheet.getRange(4, 1, 1, config.headers.length);
    headerTextRange.setValues([config.headers]);
    headerTextRange.setFontWeight("bold");
    headerTextRange.setFontSize(10);
  }
}

/**
 * Dynamic Subheader Block Renderer: Reads Section B2 specs from Format Dashboard.
 * Applies exact Row Height, Font Size, Font Weight, Fill Level, Alignment, and
 * Wrap Strategy dynamically per step WITHOUT hardcoded wrap overrides or dynamic timestamps.
 */
function applySubheaderBlocks_(sheet, subheaders, theme, sysName) {
  var currentRow = 5;
  var maxCols = sheet.getMaxColumns();
  var dashConfig = loadDashboardConfig_();
  var b2Specs = dashConfig.subheaderRows || [];

  subheaders.forEach(function(block) {
    var steps = [1, 2, 3, 4, 5, 6];

    steps.forEach(function(stepNum) {
      var stepSpecs = b2Specs.filter(function(s) { return Number(s.row) === stepNum; });
      var primarySpec = stepSpecs.length > 0 ? stepSpecs[0] : null;

      currentRow++;

      // 1. Row Height
      var rHeight = (primarySpec && primarySpec.height && !isNaN(primarySpec.height)) ? Number(primarySpec.height) : 25;
      sheet.setRowHeight(currentRow, rHeight);

      // 2. Background Fill Level
      if (primarySpec && primarySpec.fillLevel) {
        var fillLevelKey = String(primarySpec.fillLevel || "").toLowerCase().replace(/[^a-z0-9]/g, "");
        var bgColor = (fillLevelKey && fillLevelKey !== "none") ? theme[fillLevelKey] : null;
        if (bgColor) {
          sheet.getRange(currentRow, 1, 1, maxCols).setBackground(bgColor);
        }
      }

      // 3. Step-Specific Content & Formatting
      if (stepNum === 1 || stepNum === 5) {
        return; // Blank Buffer Rows
      }

      if (stepNum === 2) {
        stepSpecs.forEach(function(spec) {
          var isTimestamp = String(spec.purpose || "").toUpperCase().indexOf("TIMESTAMP") !== -1 || spec.targetCell === "C" || spec.targetCell === "Col C";
          if (!isTimestamp) {
            var cellA = sheet.getRange(currentRow, 1);
            cellA.setValue(block.title);
            if (spec.fontSize) cellA.setFontSize(Number(spec.fontSize));
            if (spec.fontWeight) cellA.setFontWeight(spec.fontWeight.toLowerCase() === "bold" ? "bold" : "normal");
            if (spec.wrap) cellA.setWrapStrategy(getWrapStrategyEnum_(spec.wrap));
            if (spec.alignment) cellA.setHorizontalAlignment(spec.alignment.toLowerCase());
          } else {
            if (sysName.indexOf("Timing") !== -1 || sysName.indexOf("Quality") !== -1 || sysName.indexOf("Change") !== -1 || block.stampLastUpdated) {
              var cellC = sheet.getRange(currentRow, 3);
              cellC.setValue("Last Updated");
              if (spec.fontSize) cellC.setFontSize(Number(spec.fontSize));
              if (spec.fontWeight) cellC.setFontStyle(spec.fontWeight.toLowerCase() === "italic" ? "italic" : "normal");
              if (spec.wrap) cellC.setWrapStrategy(getWrapStrategyEnum_(spec.wrap));
              if (spec.alignment) cellC.setHorizontalAlignment(spec.alignment.toLowerCase());
            }
          }
        });
        return;
      }

      if (stepNum === 3) {
        return; // Accent Spacer Row
      }

      if (stepNum === 4) {
        var wrapStrat = (primarySpec && primarySpec.wrap) ? getWrapStrategyEnum_(primarySpec.wrap) : SpreadsheetApp.WrapStrategy.WRAP;
        if (block.headers && block.headers.length > 0) {
          var headerTextRange = sheet.getRange(currentRow, 1, 1, block.headers.length);
          headerTextRange.setValues([block.headers]);
          if (primarySpec && primarySpec.fontSize) headerTextRange.setFontSize(Number(primarySpec.fontSize));
          if (primarySpec && primarySpec.fontWeight) headerTextRange.setFontWeight(primarySpec.fontWeight.toLowerCase() === "bold" ? "bold" : "normal");
          if (primarySpec && primarySpec.alignment) headerTextRange.setHorizontalAlignment(primarySpec.alignment.toLowerCase());
          headerTextRange.setWrapStrategy(wrapStrat);
        }
        return;
      }

      if (stepNum === 6) {
        var wrapStrat6 = (primarySpec && primarySpec.wrap) ? getWrapStrategyEnum_(primarySpec.wrap) : SpreadsheetApp.WrapStrategy.CLIP;
        var placeholderCell = sheet.getRange(currentRow, 1);
        placeholderCell.setValue("Insert new data here");
        if (primarySpec && primarySpec.fontSize) placeholderCell.setFontSize(Number(primarySpec.fontSize));
        if (primarySpec && primarySpec.fontWeight) placeholderCell.setFontWeight(primarySpec.fontWeight.toLowerCase() === "bold" ? "bold" : "normal");
        if (primarySpec && primarySpec.alignment) placeholderCell.setHorizontalAlignment(primarySpec.alignment.toLowerCase());
        placeholderCell.setWrapStrategy(wrapStrat6);
        return;
      }
    });
  });
}

function getWrapStrategyEnum_(wrapStr) {
  var w = String(wrapStr || "").toUpperCase();
  if (w === "WRAP") return SpreadsheetApp.WrapStrategy.WRAP;
  if (w === "OVERFLOW") return SpreadsheetApp.WrapStrategy.OVERFLOW;
  return SpreadsheetApp.WrapStrategy.CLIP;
}

function applyColumnWidthsEngine_(sheet, config) {
  if (!sheet) return;
  var targetCols = sheet.getMaxColumns();
  var dashConfig = loadDashboardConfig_();
  var defaultColWidth = Number(dashConfig.globals["Default Column Width"]) || GLOBAL_DEFAULTS.DEFAULT_COL_WIDTH;
  var widthSpec = String(config.columnWidths || "").trim();

  if (widthSpec && widthSpec.toUpperCase() !== "FIT_TO_DATA") {
    var parsedWidths = widthSpec.split(",").map(item => parseInt(item.trim(), 10));
    if (parsedWidths.some(w => !isNaN(w) && w > 0)) {
      if (parsedWidths.length === 1 && !isNaN(parsedWidths[0]) && parsedWidths[0] > 0) {
        for (var col = 1; col <= targetCols; col++) {
          sheet.setColumnWidth(col, parsedWidths[0]);
        }
      } else {
        for (var i = 0; i < targetCols; i++) {
          var colWidth = (i < parsedWidths.length && !isNaN(parsedWidths[i]) && parsedWidths[i] > 0) 
                         ? parsedWidths[i] 
                         : defaultColWidth;
          sheet.setColumnWidth(i + 1, colWidth);
        }
      }
      return;
    }
  }

  for (var c = 1; c <= targetCols; c++) {
    sheet.setColumnWidth(c, defaultColWidth);
  }
}

// ============================================================================

function applyOperationalFormatting_(templateSheet, systemSheetName) {
  if (!templateSheet) return;
  const config = loadDashboardConfig_();
  
  const headers = config.sheetHeaders[systemSheetName] || 
                  config.sheetHeaders[getStandardSheetKey_(systemSheetName)] || [];
  
  if (!headers || headers.length === 0) return;

  resizeSheetGrid_(templateSheet, templateSheet.getMaxRows(), headers.length);
  
  const headerRange = templateSheet.getRange(4, 1, 1, headers.length);
  headerRange.setValues([headers]);
  headerRange.setFontWeight("bold");

  const colDefs = config.columnDefinitions || {};
  const defaultWidth = Number(config.globals["Default Column Width"]) || GLOBAL_DEFAULTS.DEFAULT_COL_WIDTH;
  const maxRows = templateSheet.getMaxRows();

  for (let c = 0; c < headers.length; c++) {
    const headerName = String(headers[c] || "").trim();
    const colDef = colDefs[headerName] || colDefs[headerName.toLowerCase()] || {};
    const colIdx = c + 1;

    if (colDef.width && colDef.width > 0) {
      templateSheet.setColumnWidth(colIdx, colDef.width);
    } else {
      templateSheet.setColumnWidth(colIdx, defaultWidth);
    }

    if (maxRows >= 5) {
      const dataColRange = templateSheet.getRange(5, colIdx, maxRows - 4, 1);
      
      if (colDef.format) {
        dataColRange.setNumberFormat(colDef.format);
      } else if (colDef.dateColumn) {
        dataColRange.setNumberFormat("mm/dd/yyyy");
      }

      if (colDef.align) {
        dataColRange.setHorizontalAlignment(colDef.align);
      }
      if (colDef.vAlign) {
        dataColRange.setVerticalAlignment(colDef.vAlign);
      }
      if (colDef.wrap) {
        dataColRange.setWrapStrategy(
          colDef.wrap === "WRAP" ? SpreadsheetApp.WrapStrategy.WRAP :
          colDef.wrap === "OVERFLOW" ? SpreadsheetApp.WrapStrategy.OVERFLOW :
          SpreadsheetApp.WrapStrategy.CLIP
        );
      }
    }
  }
}

function getStandardSheetKey_(sysName) {

    "Raw Data": "Raw Data", "Refined Data": "Refined Data",
    "Disenrolled": "Disenrolled Exclusion", "Master List": "Master List",
    "Monthly Change": "Monthly Change"
  };
  return map[sysName] || sysName;
}

function getTemplateConfigFromDashboard_(systemSheetName) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dashSheet = ss.getSheetByName("Format Dashboard") || ss.getSheetByName("Default - Format Dashboard");
  
  if (!dashSheet) {
    dashSheet = buildFormatDashboardTemplate_();
  }

  const config = loadDashboardConfig_();
  const targetLower = String(systemSheetName || "").toLowerCase().trim();

  let sheetDef = (config.sheetDefinitions || []).find(d => {
    const dName = String(d.systemSheetName || "").toLowerCase().trim();
    const dTitle = String(d.reportTitle || "").toLowerCase().trim();
    return dName === targetLower || 
           dTitle === targetLower ||
           dName.indexOf(targetLower) !== -1 || 
           targetLower.indexOf(dName) !== -1 ||
           dTitle.indexOf(targetLower) !== -1 ||
           targetLower.indexOf(dTitle) !== -1;
  });

  if (!sheetDef) {
    sheetDef = {
      systemSheetName: systemSheetName,
      reportTitle: systemSheetName,
      rank: 999,
      columnCount: 10,
      columnWidths: "105",
      usesSubheaders: (targetLower.indexOf("timing") !== -1 || targetLower.indexOf("quality") !== -1 || targetLower.indexOf("change") !== -1),
      usesAlternatingColors: true
    };
  }

  let headers = getHeadersForSheetType_(config, systemSheetName);
  let subheaders = getSubheadersFromDashboardConfig_(systemSheetName);

  return {
    systemSheetName: systemSheetName,
    templateName: "Template - " + systemSheetName,
    reportTitle: sheetDef.reportTitle || systemSheetName,
    sheetType: (targetLower.indexOf("timing") !== -1 || targetLower.indexOf("quality") !== -1 || targetLower === "index") ? "SYSTEM" : "DATA",
    headers: headers,
    columnCount: Math.max(sheetDef.columnCount || 0, headers ? headers.length : 0, 10),
    minimumRows: 100,
    rank: sheetDef.rank || 999,
    columnWidths: sheetDef.columnWidths || "105",
    usesSubheaders: sheetDef.usesSubheaders || false,
    usesAlternatingColors: sheetDef.usesAlternatingColors !== false,
    subheaders: subheaders
  };
}

function formatDashboardSections_(sheet) {
  if (!sheet) return;
  const theme = getTemplateTheme_("Format Dashboard");
  const maxCols = sheet.getMaxColumns();
  const values = sheet.getDataRange().getValues();

  for (let r = 0; r < values.length; r++) {
    const val = String(values[r][0] || "").trim();
    if (val.indexOf("SECTION ") === 0) {
      const bannerSheetRow = r + 1;
      const spacerSheetRow = r + 2;
      const headerSheetRow = r + 3;

      const bannerRange = sheet.getRange(bannerSheetRow, 1, 1, maxCols);
      bannerRange.setBackground(theme.level3 || "#CCE3EE");
      sheet.getRange(bannerSheetRow, 1).setFontSize(12).setFontWeight("bold");
      sheet.setRowHeight(bannerSheetRow, 25);

      if (spacerSheetRow <= sheet.getMaxRows()) {
        sheet.getRange(spacerSheetRow, 1, 1, maxCols).setBackground(theme.level1 || "#79B5D2");
        sheet.setRowHeight(spacerSheetRow, 10);
      }

      if (headerSheetRow <= sheet.getMaxRows()) {
        const headerRange = sheet.getRange(headerSheetRow, 1, 1, maxCols);
        headerRange.setBackground(theme.level2 || "#A5D0E3");
        headerRange.setFontSize(8);
        headerRange.setFontWeight("bold");
        headerRange.setWrapStrategy(SpreadsheetApp.WrapStrategy.WRAP);
        sheet.setRowHeight(headerSheetRow, 25);
      }
    }
  }
}

function appendDashboardSectionRows_(rows, sectionName, headers, dataRows, width) {
  const emptyRow = new Array(width).fill("");
  rows.push(emptyRow);

  const bannerRow = new Array(width).fill("");
  bannerRow[0] = sectionName;
  rows.push(bannerRow);

  rows.push(emptyRow);

  const headerRow = new Array(width).fill("");
  for (let i = 0; i < headers.length; i++) {
    headerRow[i] = headers[i];
  }
  rows.push(headerRow);

  rows.push(emptyRow);
  (dataRows || []).forEach(row => {
    const paddedRow = row.slice(0, width);
    while (paddedRow.length < width) paddedRow.push("");
    rows.push(paddedRow);
  });
}

function populateIndexData(targetName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheetName = targetName || "Index";
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return;

  const timestampStr = Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), "MM/dd/yyyy HH:mm:ss");
  const archiveId = PropertiesService.getDocumentProperties().getProperty("RFF_ARCHIVE_SPREADSHEET_ID") || "";

  const sectionFRules = loadSectionFTabRules_();
  const activeSheets = ss.getSheets();
  const activeRows = [];

  activeSheets.forEach(s => {
    const name = s.getName();
    if (name !== sheetName && name !== "Template - Index") {
      const visibility = s.isSheetHidden() ? "Hidden" : "Visible";
      const match = matchSheetToSectionFRule_(name, sectionFRules);
      const link = `=HYPERLINK("#gid=${s.getSheetId()}", "Open Live Tab")`;

      activeRows.push({
        name: name,
        group: match.group,
        rank: match.rank,
        link: link,
        visibility: visibility
      });
    }
  });

  activeRows.sort((a, b) => {
    if (a.rank !== b.rank) return a.rank - b.rank;
    return a.name.localeCompare(b.name);
  });

  const headers = [
    "Section/Month", "Sheet Tab Name", "Workspace Link", "Visibility",
    "", "Archive Month", "Archive Sheet Name", "Link to Sheet", "Restore Action", "Status"
  ];

  const rawRows = [
    ["Active Index", "", "", "", "", "Archive Index", "", "", "", ""],
    ["Generated", timestampStr, "", "", "", "Archive File ID", archiveId, "", "", ""],
    ["", "", "", "", "", "", "", "", "", ""],
    headers
  ];

  let currentGroupDivider = "";
  let activeIdx = 0;
  const dividerRowIndices = [];

  while (activeIdx < activeRows.length) {
    const act = activeRows[activeIdx];

    if (act && act.group !== currentGroupDivider) {
      currentGroupDivider = act.group;
      rawRows.push([currentGroupDivider, "", "", "", "", "", "", "", "", ""]);
      dividerRowIndices.push(rawRows.length);
      continue;
    }

    if (act) {
      rawRows.push(["", act.name, act.link, act.visibility, "", "", "", "", "", ""]);
      activeIdx++;
    }
  }

  resizeSheetGrid_(sheet, Math.max(rawRows.length, 20), 10);
  sheet.getRange(1, 1, rawRows.length, 10).setValues(rawRows);
  formatIndexSheet_(sheet, dividerRowIndices);
}

function formatIndexSheet_(sheet, dividerRowIndices) {
  if (!sheet) return;  
  const theme = getTemplateTheme_("Index");
  const maxGridRows = sheet.getMaxRows();
  const lastRow = Math.max(sheet.getLastRow(), 4);

  sheet.getRange(1, 1, 1, 4).setBackground(theme.level3).setFontSize(14).setFontWeight("bold");
  sheet.getRange(1, 6, 1, 5).setBackground(theme.level3).setFontSize(14).setFontWeight("bold");
  
  sheet.getRange(2, 1, 1, 4).setBackground(theme.level3).setFontSize(10).setFontWeight("normal");
  sheet.getRange(2, 6, 1, 5).setBackground(theme.level3).setFontSize(10).setFontWeight("normal");

  sheet.getRange(3, 1, 1, 10).setBackground(theme.level1).setFontSize(10).setFontWeight("normal");

  sheet.getRange(4, 1, 1, 4).setBackground(theme.level2).setFontSize(10).setFontWeight("bold");
  sheet.getRange(4, 6, 1, 5).setBackground(theme.level2).setFontSize(10).setFontWeight("bold");

  if (lastRow >= 5) {
    applyNativeBandingSafe_(sheet, 5, 10, lastRow, theme.level4, theme.level5);
  }

  sheet.getRange(1, 5, maxGridRows, 1).setBackground(theme.level1);

  if (dividerRowIndices && dividerRowIndices.length > 0) {
    dividerRowIndices.forEach(rIdx => {
      if (rIdx >= 5 && rIdx <= lastRow) {
        sheet.getRange(rIdx, 1, 1, 4).setBackground(theme.level2);
        sheet.getRange(rIdx, 1).setFontWeight("bold");
      }
    });
  }

  applyColumnWidths_(sheet, "160, 185, 160, 160, 30, 160, 185, 160, 160, 160");
  sheet.setFrozenRows(4);
  sheet.setFrozenColumns(1);
}

function writeSubheaderSectionData_(sheet, sectionTitle, dataMatrix) {
  if (!sheet || !dataMatrix || dataMatrix.length === 0) return;

  const values = sheet.getDataRange().getValues();
  let anchorRow = -1;

  for (let r = 0; r < values.length; r++) {
    if (String(values[r][0] || "").trim().toUpperCase().indexOf(sectionTitle.toUpperCase()) !== -1) {
      anchorRow = r + 1;
      break;
    }
  }

  if (anchorRow === -1) return;

  const dataStartRow = anchorRow + 4;
  const numNewRows = dataMatrix.length;

  let nextSectionRow = values.length + 1;
  for (let r = anchorRow; r < values.length; r++) {
    const text = String(values[r][0] || "").trim().toUpperCase();
    if (r + 1 > anchorRow && text.indexOf("SECTION ") === 0) {
      nextSectionRow = r + 1;
      break;
    }
  }

  const availableDataRows = Math.max(0, nextSectionRow - dataStartRow - 1);
  if (numNewRows > availableDataRows) {
    const rowsToInsert = numNewRows - availableDataRows;
    sheet.insertRowsAfter(dataStartRow, rowsToInsert);
  }

  const normalizedMatrix = dataMatrix.map(row => {
    const out = row.slice(0, sheet.getMaxColumns());
    while (out.length < 4) out.push("");
    return out;
  });

  sheet.getRange(dataStartRow, 1, normalizedMatrix.length, normalizedMatrix[0].length).setValues(normalizedMatrix);
}

function loadDashboardConfig_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Format Dashboard") || ss.getSheetByName("Default - Format Dashboard");
  if (!sheet) return getFallbackDashboardConfig_();

  const values = sheet.getDataRange().getValues();
  const config = { 
    globals: {}, 
    titleRows: [], 
    subheaderRows: [], 
    sheetDefinitions: [], 
    columnDefinitions: {}, 
    sheetHeaders: {} 
  };
  const dashIndex = buildDashboardSectionIndex_(sheet);

  // Section A - Global Settings
  const secA = dashIndex["SECTION A - GLOBAL SETTINGS"];
  if (secA) {
    for (let r = secA.startRow; r <= secA.endRow; r++) {
      const key = String(values[r][0] || "").trim();
      if (key) config.globals[key] = values[r][1];
    }
  }

  // Section B - Title Rows
  const secB1 = dashIndex["SECTION B - TITLE ROWS"] || dashIndex["SECTION B1 - TITLE ROWS"] || dashIndex["SECTION B1"];
  if (secB1) {
    for (let r = secB1.startRow; r <= secB1.endRow; r++) {
      const sheetType = String(values[r][0] || "").trim();
      if (sheetType && sheetType.indexOf("SECTION ") === -1) {
        config.titleRows.push({
          sheetType: sheetType,
          row: Number(values[r][1]) || (r - secB1.startRow + 1),
          purpose: String(values[r][2] || "").trim(),
          valueSource: String(values[r][3] || "").trim(),
          label: String(values[r][4] || "").trim(),
          targetCell: String(values[r][5] || "").trim(),
          height: Number(values[r][6]) || 25,
          fontSize: Number(values[r][7]) || 10,
          fontWeight: String(values[r][8] || "Normal").trim(),
          fillLevel: String(values[r][9] || "Level 3").trim(),
          alignment: String(values[r][10] || "Left").trim(),
          wrap: String(values[r][11] || "Clip").trim()
        });
      }
    }
  }

  // Section B2 - Subheader Rows (13-Column Structure Aligned with B1)
  const secB2 = dashIndex["SECTION B2 - SUBHEADER ROWS"] || dashIndex["SECTION B2"];
  if (secB2) {
    for (let r = secB2.startRow; r <= secB2.endRow; r++) {
      const sheetType = String(values[r][0] || "").trim();
      if (sheetType && sheetType.indexOf("SECTION ") === -1) {
        config.subheaderRows.push({
          sheetType: sheetType,
          row: Number(values[r][1]) || (r - secB2.startRow + 1),
          purpose: String(values[r][2] || "").trim(),
          valueSource: String(values[r][3] || "").trim(),
          label: String(values[r][4] || "").trim(),
          targetCell: String(values[r][5] || "").trim(),
          height: Number(values[r][6]) || 25,
          fontSize: Number(values[r][7]) || 10,
          fontWeight: String(values[r][8] || "Normal").trim(),
          fillLevel: String(values[r][9] || "None").trim(),
          alignment: String(values[r][10] || "Left").trim(),
          wrap: String(values[r][11] || "Clip").trim()
        });
      }
    }
  }

  // Section C - Sheet Definitions
  const secC = dashIndex["SECTION C -  Template Structural Specs"] || dashIndex["SECTION C - SHEET DEFINITIONS"] || dashIndex["SECTION C"];
  if (secC) {
    for (let r = secC.startRow; r <= secC.endRow; r++) {
      const sysName = String(values[r][0] || "").trim();
      if (sysName && sysName.indexOf("SECTION ") === -1) {
        config.sheetDefinitions.push({
          systemSheetName: sysName,
          group: String(values[r][1] || "Template").trim(),
          reportTitle: String(values[r][2] || sysName).trim(),
          rank: Number(values[r][3]) || 999,
          baseColor: String(values[r][4] || "#79B5D2").trim(),
          columnCount: Number(values[r][11]) || 10,
          columnWidths: String(values[r][12] || "105").trim(),
          usesAlternatingColors: values[r][16] !== false,
          usesSubheaders: values[r][17] === true || String(values[r][17]).toUpperCase() === "TRUE"
        });
      }
    }
  }

  // Section E - System Sheet Surfaces
  const secE = dashIndex["SECTION E - Source/Unfomatted"] || dashIndex["SECTION E - SYSTEM SHEET SURFACES"];
  if (secE) {
    for (let r = secE.startRow; r <= secE.endRow; r++) {
      const sysName = String(values[r][0] || "").trim();
      if (sysName && sysName.indexOf("SECTION ") === -1) {
        config.sheetDefinitions.push({
          systemSheetName: sysName,
          reportTitle: String(values[r][1] || sysName).trim(),
          rank: Number(values[r][3]) || 500,
          baseColor: String(values[r][4] || "#79B5D2").trim(),
          columnCount: 10,
          columnWidths: (sysName.indexOf("Timing") !== -1) ? "250, 250, 500, 350, 250, 500" : (sysName.indexOf("Quality") !== -1) ? "250, 250, 350, 300" : "105",
          usesAlternatingColors: true,
          usesSubheaders: (sysName.indexOf("Timing") !== -1 || sysName.indexOf("Quality") !== -1)
        });
      }
    }
  }

  // Section F - Column Definitions
  const secF = dashIndex["SECTION F - COLUMN DEFINITIONS"] || dashIndex["SECTION F"];
  if (secF) {
    for (let r = secF.startRow; r <= secF.endRow; r++) {
      const header = String(values[r][0] || "").trim();
      if (header) {
        const colData = {
          width: Number(values[r][1]) || 105,
          fontSize: Number(values[r][2]) || 10,
          dateColumn: values[r][3] === true || String(values[r][3]).toUpperCase() === "TRUE",
          hideColumn: values[r][4] === true || String(values[r][4]).toUpperCase() === "TRUE",
          wrap: String(values[r][5] || "CLIP").trim().toUpperCase(),
          align: String(values[r][6] || "left").trim().toLowerCase(),
          vAlign: String(values[r][7] || "middle").trim().toLowerCase(),
          format: String(values[r][8] || "").trim()
        };
        config.columnDefinitions[header] = colData;
        config.columnDefinitions[header.toLowerCase()] = colData;
      }
    }
  }

  // Section G - Sheet Headers
  const secG = dashIndex["SECTION G - SHEET HEADERS"] || dashIndex["SECTION G"];
  if (secG) {
    for (let r = secG.startRow; r <= secG.endRow; r++) {
      const sheetType = String(values[r][0] || "").trim();
      const header = String(values[r][2] || "").trim();
      if (sheetType && header) {
        if (!config.sheetHeaders[sheetType]) config.sheetHeaders[sheetType] = [];
        config.sheetHeaders[sheetType].push(header);
      }
    }
  }

  return config;
}

function buildDashboardSectionIndex_(dashboardSheet) {
  const values = dashboardSheet.getDataRange().getValues();
  const index = {};
  let currentSection = null;
  for (let r = 0; r < values.length; r++) {
    const val = String(values[r][0] || "").trim();
    if (val.indexOf("SECTION ") === 0) {
      if (currentSection) index[currentSection].endRow = r - 1;
      currentSection = val;
      index[currentSection] = { startRow: r + 2, endRow: values.length - 1 };
    }
  }
  return index;
}

function getTemplateTheme_(systemSheetName) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var dashSheet = ss.getSheetByName("Format Dashboard") || ss.getSheetByName("Default - Format Dashboard");
    if (!dashSheet) return getFallbackTheme_();
    var data = dashSheet.getDataRange().getValues();
    var hslPercents = getHslPercentsFromDashboard_(dashSheet);

    for (var i = 0; i < data.length; i++) {
      var sysKey = String(data[i][0] || "").trim();
      if (sysKey === systemSheetName || sysKey.toLowerCase() === String(systemSheetName || "").toLowerCase()) {
        var rawBase = String(data[i][4] || data[i][3] || "#79B5D2").trim();
        var baseHex = isValidHex_(rawBase) ? rawBase : "#79B5D2";
        var computed = calculateThemeLevels_(baseHex, hslPercents);

        var l1 = String(data[i][5] || "").trim();
        var l2 = String(data[i][6] || "").trim();
        var l3 = String(data[i][7] || "").trim();
        var l4 = String(data[i][8] || "").trim();
        var l5 = String(data[i][9] || "").trim();
        var fontCol = String(data[i][10] || "").trim();

        return {
          baseColor: baseHex,
          level1: isValidHex_(l1) ? l1 : computed.level1,
          level2: isValidHex_(l2) ? l2 : computed.level2,
          level3: isValidHex_(l3) ? l3 : computed.level3,
          level4: isValidHex_(l4) ? l4 : computed.level4,
          level5: isValidHex_(l5) ? l5 : computed.level5,
          titleFontColor: isValidHex_(fontCol) ? fontCol : GLOBAL_DEFAULTS.STANDARD_FONT_COLOR
        };
      }
    }
  } catch (e) {}
  return getFallbackTheme_();
}

function getHslPercentsFromDashboard_(dashSheet) {
  var percents = { l1: 60, l2: 75, l3: 85, l4: 97, l5: 90 };
  if (!dashSheet) return percents;
  try {
    var data = dashSheet.getRange("A1:B40").getValues();
    for (var i = 0; i < data.length; i++) {
      var key = String(data[i][0] || "").trim();
      var val = Number(data[i][1]);
      if (key === "HSL Level 1 Lightness %" && !isNaN(val)) percents.l1 = val;
      if (key === "HSL Level 2 Lightness %" && !isNaN(val)) percents.l2 = val;
      if (key === "HSL Level 3 Lightness %" && !isNaN(val)) percents.l3 = val;
      if (key === "HSL Level 4 Lightness %" && !isNaN(val)) percents.l4 = val;
      if (key === "HSL Level 5 Lightness %" && !isNaN(val)) percents.l5 = val;
    }
  } catch (e) {}
  return percents;
}

function recalculateDashboardHexCodes_(dashboardSheet) {
  if (!dashboardSheet) return;
  const rawDashData = dashboardSheet.getDataRange().getValues();
  const hslPercents = getHslPercentsFromDashboard_(dashboardSheet);

  for (let r = 0; r < rawDashData.length; r++) {
    const firstCell = String(rawDashData[r][0] || "").trim();

    if (!firstCell || firstCell.indexOf("SECTION ") === 0 || firstCell === "System Sheet Name" || firstCell === "Sheet Type" || firstCell === "Setting" || firstCell === "Header") {
      continue;
    }

    const potentialHex = String(rawDashData[r][4] || rawDashData[r][3] || "").trim();
    if (isValidHex_(potentialHex)) {
      const theme = calculateThemeLevels_(potentialHex, hslPercents);
      const targetRange = dashboardSheet.getRange(r + 1, 6, 1, 5);
      targetRange.clearNote();
      targetRange.setValues([[theme.level1, theme.level2, theme.level3, theme.level4, theme.level5]]);
    }
  }
}

  else if (Array.isArray(widthSpec)) widths = widthSpec;
  for (var i = 0; i < widths.length; i++) {
    if (!isNaN(widths[i]) && widths[i] > 0) sheet.setColumnWidth(i + 1, widths[i]);
  }
}

function applyNativeBandingSafe_(sheet, startRow, totalCols, totalRows, color1, color2) {
  if (!sheet || totalRows < startRow) return;
  var cols = totalCols || sheet.getMaxColumns();
  var numDataRows = totalRows - startRow + 1;

  var c1 = isValidHex_(color1) ? color1 : "#eff8cf";
  var c2 = isValidHex_(color2) ? color2 : "#f3fadd";

  try {
    var existingBandings = sheet.getBandings();
    for (var b = 0; b < existingBandings.length; b++) {
      try { existingBandings[b].remove(); } catch(e) {}
    }

    if (numDataRows > 0) {
      var targetRange = sheet.getRange(startRow, 1, numDataRows, cols);
      var banding = targetRange.applyRowBanding(SpreadsheetApp.BandingTheme.CUSTOM, false, false);
      banding.setFirstRowColor(c1);
      banding.setSecondRowColor(c2);
    }
  } catch (e) {
    if (numDataRows > 0) {
      var bgColors = [];
      for (var r = 0; r < numDataRows; r++) {
        var rowColor = (r % 2 === 0) ? c1 : c2;
        bgColors.push(new Array(cols).fill(rowColor));
      }
      sheet.getRange(startRow, 1, numDataRows, cols).setBackgrounds(bgColors);
    }
  }
}

function positionSheetBySectionFRank_(sheet, rank) {
  if (!sheet) return;
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var currentSheet = ss.getSheetByName(sheet.getName());
    if (!currentSheet) return;

    var targetRank = rank || getTargetRankForSheet_(currentSheet.getName());
    var sheets = ss.getSheets();
    
    var targetPos = 1;
    for (var i = 0; i < sheets.length; i++) {
      if (sheets[i].getName() === currentSheet.getName()) continue;
      var otherRank = getTargetRankForSheet_(sheets[i].getName());
      if (otherRank < targetRank) {
        targetPos++;
      }
    }

    ss.setActiveSheet(currentSheet);
    ss.moveActiveTab(Math.min(Math.max(1, targetPos), sheets.length));
  } catch (e) {}
}

function getTargetRankForSheet_(sheetName) {
  const rules = loadSectionFTabRules_();
  const match = matchSheetToSectionFRule_(sheetName, rules);
  return match.rank;
}

function matchSheetToSectionFRule_(sheetName, rules) {
  const cleanName = String(sheetName || "").trim().toLowerCase().replace(/[^a-z0-9]/g, "");
  for (let i = 0; i < rules.length; i++) {
    const ruleClean = rules[i].prefix.toLowerCase().replace(/[^a-z0-9]/g, "");
    if (cleanName === ruleClean || cleanName.indexOf(ruleClean) === 0) return rules[i];
  }
  if (sheetName.indexOf("Template - ") === 0 || sheetName === "RFF_BASE_TEMPLATE") return { group: "Template", rank: 800 };
  return { group: "Unformatted", rank: 999 };
}

function resizeSheetGrid_(sheet, targetRows, targetCols) {
  var currentRows = sheet.getMaxRows(), currentCols = sheet.getMaxColumns();
  var frozenCols = sheet.getFrozenColumns();
  var frozenRows = sheet.getFrozenRows();

  if (targetCols <= frozenCols && frozenCols > 0) {
    sheet.setFrozenColumns(Math.max(0, targetCols - 1));
  }
  if (targetRows <= frozenRows && frozenRows > 0) {
    sheet.setFrozenRows(Math.max(0, targetRows - 1));
  }

  if (currentRows < targetRows) sheet.insertRowsAfter(currentRows, targetRows - currentRows);
  else if (currentRows > targetRows) sheet.deleteRows(targetRows + 1, currentRows - targetRows);

  if (currentCols < targetCols) sheet.insertColumnsAfter(currentCols, targetCols - currentCols);
  else if (currentCols > targetCols) sheet.deleteColumns(targetCols + 1, currentCols - targetCols);
}

function checkTemplateFormatSignature_(templateName, config) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(templateName); if (!sheet) return false;
  var currentHash = computeConfigHash_(config); var metadata = sheet.getDeveloperMetadata();
  for (var i = 0; i < metadata.length; i++) { if (metadata[i].getKey() === "FORMAT_HASH") return metadata[i].getValue() === currentHash; }
  return false;
}

}

function computeConfigHash_(config) {
  var digest = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, JSON.stringify(config));
  return digest.map(byte => (byte < 0 ? byte + 256 : byte).toString(16)).join("");
}

function isValidHex_(hexStr) { return /^#[0-9A-F]{6}$/i.test(String(hexStr || "").trim()); }

function hexToHsl_(hex) {
  let c = String(hex || "").replace('#', '').trim();
  if (c.length === 3) c = c.split('').map(x => x + x).join('');
  if (c.length !== 6) return { h: 200, s: 50, l: 50 };
  const r = parseInt(c.substring(0, 2), 16) / 255, g = parseInt(c.substring(2, 4), 16) / 255, b = parseInt(c.substring(4, 6), 16) / 255;
  const max = Math.max(r, g, b), min = Math.min(r, g, b);
  let h = 0, s = 0, l = (max + min) / 2;
  if (max !== min) {
    const d = max - min; s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
    switch (max) { case r: h = (g - b) / d + (g < b ? 6 : 0); break; case g: h = (b - r) / d + 2; break; case b: h = (r - g) / d + 4; break; }
    h /= 6;
  }
  return { h: Math.round(h * 360), s: Math.round(s * 100), l: Math.round(l * 100) };
}

function hslToHex_(h, s, l) {
  l /= 100; const a = (s * Math.min(l, 1 - l)) / 100;
  const f = n => { const k = (n + h / 30) % 12; const color = l - a * Math.max(Math.min(k - 3, 9 - k, 1), -1); return Math.round(255 * color).toString(16).padStart(2, '0'); };
  return `#${f(0)}${f(8)}${f(4)}`.toUpperCase();
}

function calculateThemeLevels_(baseHex, hslPercents) {
  const hsl = hexToHsl_(baseHex);
  return {
    base: baseHex,
    level1: hslToHex_(hsl.h, hsl.s, hslPercents.l1 || 60),
    level2: hslToHex_(hsl.h, hsl.s, hslPercents.l2 || 75),
    level3: hslToHex_(hsl.h, hsl.s, hslPercents.l3 || 85),
    level4: hslToHex_(hsl.h, hsl.s, hslPercents.l4 || 97),
    level5: hslToHex_(hsl.h, hsl.s, hslPercents.l5 || 90)
  };
}
